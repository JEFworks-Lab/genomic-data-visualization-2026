### Question and description of solution
QUESTION: If I perform non-linear dimensionality reduction on PCs, what happens when I vary how many PCs should I use?

I normalized the Visium count matrix using CPM followed by a log10 transform, selected the top 1000 highly variable genes by variance, 
and performed PCA. I then ran tSNE on the first k PCs (k = 5, 10, 20, 50) and animated the resulting embeddings to show how the apparent cluster 
structure changes as more PCs (increasingly low-variance components) are included. Spots are colored by k-means clusters computed in PCA space (k=6, chosen using an elbow plot), 
demonstrating that the embedding becomes more/less stable and cluster separation changes as the input dimensionality increases.

```r


# Extra Credit HW1: Animated visualization with gganimate
# Q4: Vary number of PCs before tSNE

# install.packages(c("ggplot2","dplyr","Rtsne","gganimate","gifski","viridis"))

suppressPackageStartupMessages({
  library(ggplot2)
  library(dplyr)
  library(Rtsne)
  library(gganimate)
  library(gifski)
  library(viridis)
})

set.seed(42)

# Loading data
infile <- "/Users/pmusenge/Desktop/genomic_data_vis/genomic-data-visualization-2026/data/Visium-IRI-ShamR_matrix.csv.gz"
data <- read.csv(infile, check.names = FALSE)

cat("Data dim:", dim(data), "\n")
cat("First 12 colnames:\n")
print(colnames(data)[1:12])

# schema and checks
spot_id <- data[[1]]
if (any(is.na(spot_id)) || anyDuplicated(spot_id) > 0) {
  warning("spot_id has NA and/or duplicates. Replacing with sequential IDs.")
  spot_id <- paste0("spot_", seq_len(nrow(data)))
}

pos <- data.frame(x = data[["x"]], y = data[["y"]])
rownames(pos) <- spot_id

gexp <- as.matrix(data[, 4:ncol(data), drop = FALSE])
rownames(gexp) <- spot_id

cat("pos dim:", dim(pos), "\n")
cat("gexp dim:", dim(gexp), "\n")

# Normalization (CPM + log10) 
totgexp <- rowSums(gexp)
mat <- log10(sweep(gexp, 1, totgexp, "/") * 1e6 + 1)

# Highly variable genes 
gene_variances <- apply(mat, 2, var)
n_hvg <- min(1000, length(gene_variances))
hvg <- names(sort(gene_variances, decreasing = TRUE))[1:n_hvg]
mat_hvg <- mat[, hvg, drop = FALSE]
cat("HVGs used:", length(hvg), "\n")

# PCA 
pcs <- prcomp(mat_hvg, center = TRUE, scale. = TRUE)

max_k <- min(50, ncol(pcs$x))
pc_scores <- pcs$x[, 1:max_k, drop = FALSE]
cat("Max PCs available/used:", max_k, "\n")


# Scree plot
pve <- pcs$sdev^2 / sum(pcs$sdev^2)
cum_pve <- cumsum(pve)

scree_df <- data.frame(
  PC = seq_along(pve),
  PVE = pve,
  Cumulative = cum_pve
)

p_scree <- ggplot(scree_df[1:max_k, ], aes(x = PC, y = PVE)) +
  geom_point() +
  geom_line() +
  labs(
    title = "Scree plot (PCA on HVGs)",
    subtitle = "Proportion variance explained (PVE) per PC",
    x = "Principal Component",
    y = "PVE"
  ) +
  theme_minimal(base_size = 12)

print(p_scree)
ggsave("final_visium_scree_plot.png", p_scree, width = 7, height = 4, dpi = 300)

# Elbow plot for K-means
set.seed(42)
k_grid <- 2:12
wss <- sapply(k_grid, function(k) {
  kmeans(pc_scores, centers = k, nstart = 25)$tot.withinss
})

elbow_df <- data.frame(k = k_grid, tot_withinss = wss)

p_elbow <- ggplot(elbow_df, aes(x = k, y = tot_withinss)) +
  geom_point() +
  geom_line() +
  labs(
    title = "K-means elbow plot (on PC space)",
    subtitle = "Total within-cluster sum of squares (WSS)",
    x = "Number of clusters (k)",
    y = "Total within-cluster SS"
  ) +
  theme_minimal(base_size = 12)

print(p_elbow)
ggsave("final_visium_kmeans_elbow.png", p_elbow, width = 7, height = 4, dpi = 300)


# Kmeans on PCs (w/ coloring)
# k=6 chosen based on elbow plot: diminishing returns beyond ~5–6 clusters
k_clusters <- 6
km <- kmeans(pc_scores, centers = k_clusters, nstart = 25)
cluster <- factor(km$cluster)

# tSNE sweep over number of PCs 
pc_grid <- c(5, 10, 20, 50)
pc_grid <- pc_grid[pc_grid <= max_k]
if (length(pc_grid) == 0) stop("No PC values available (max_k too small).")

#AI Help; Prompt: what does the error below on tSNE perplexity mean?
n <- nrow(pc_scores)
# perplexity must be < (n-1)/3
max_perp <- floor((n - 1) / 3)
perplexity <- min(30, max_perp - 1)
if (perplexity < 5) perplexity <- 5
cat("Using tSNE perplexity =", perplexity, "with n =", n, "\n")

tsne_list <- lapply(pc_grid, function(k) {
  set.seed(100 + k)
  Xk <- pc_scores[, 1:k, drop = FALSE]
  
  fit <- Rtsne(
    Xk,
    dims = 2,
    perplexity = perplexity,
    theta = 0.5,
    max_iter = 1000,
    verbose = TRUE,
    check_duplicates = FALSE
  )
  
  data.frame(
    spot_id = rownames(Xk),
    tsne1 = fit$Y[, 1],
    tsne2 = fit$Y[, 2],
    pc_k = paste0("PCs = ", k),
    stringsAsFactors = FALSE
  )
})

tsne_df <- bind_rows(tsne_list) %>%
  mutate(
    pc_k = factor(pc_k, levels = paste0("PCs = ", pc_grid))
  ) %>%
  left_join(
    data.frame(
      spot_id = rownames(pos),
      x = pos$x,
      y = pos$y,
      tot = totgexp,
      kmeans = cluster,
      stringsAsFactors = FALSE
    ),
    by = "spot_id"
  )

# Animated plot
# Fixing axis limits so the animation doesn't "jump/zoom"
xlim_all <- range(tsne_df$tsne1, finite = TRUE)
ylim_all <- range(tsne_df$tsne2, finite = TRUE)

p_anim <- ggplot(tsne_df, aes(tsne1, tsne2, color = kmeans)) +
  geom_point(size = 0.9, alpha = 0.85) +
  coord_fixed() +
  scale_color_viridis_d(option = "D", end = 0.95) +
  xlim(xlim_all) +
  ylim(ylim_all) +
  labs(
    title = "Visium spots: tSNE using {closest_state}",
    subtitle = paste0("CPM+log10 normalization; HVGs=", n_hvg,
                      "; Kmeans(k=", k_clusters, ") on PCs; perplexity=", perplexity),
    x = "tSNE 1", y = "tSNE 2", color = "Kmeans"
  ) +
  theme_minimal(base_size = 12) +
  theme(plot.title = element_text(face = "bold")) +
  transition_states(pc_k, transition_length = 2, state_length = 1) +
  ease_aes("cubic-in-out")

# Render GIF
out_gif <- "pmuseng1_visium_tsne_pc_sweep.gif"

anim <- animate(
  p_anim,
  fps = 12,
  width = 900,
  height = 700,
  renderer = gifski_renderer(out_gif)
)
anim
anim_save(out_gif, animation = anim)
cat("Saved GIF:", out_gif, "\n")



```
