To identify the tissue structure in the CODEX spleen dataset, I performed quality control by removing low-signal cells and applied an arcsinh transformation to normalize protein expression. I then used PCA to capture the main biological variation and selected the number of principal components based on cumulative variance explained. tSNE visualization showed clearly separated cell populations, and k-means clustering with the number of clusters chosen using silhouette analysis identified stable cellular groups. When mapped back to spatial coordinates, clusters formed organized regions rather than random distributions. Differential expression and module scoring revealed two dominant immune populations: a CD3e-enriched cluster corresponding to T lymphocytes (T cells) and a CD20-enriched cluster corresponding to B lymphocytes (B cells). Spatially, T cells formed a broader peri-follicular region, while B cells appeared in localized follicle-like patches. This organization is consistent with known splenic white pulp anatomy, where T cells occupy periarteriolar lymphoid sheath (PALS) regions surrounding central arterioles, and B cells form lymphoid follicles adjacent to these T-cell zones. The binned spatial cluster visualization further highlighted a lymphocyte-dense compartment consistent with follicular organization. Statistical testing confirmed significant enrichment of marker expression within these clusters (Wilcoxon p < 2.2e-16). Together, the clustering, marker expression, and spatial organization support the conclusion that the CODEX dataset represents white pulp tissue containing both T-cell and B-cell regions. While the spatial boundaries are somewhat diffuse due to segmentation noise and the limited field of view typical of multiplex imaging data, the consistent marker enrichment and regional organization provide strong evidence for this interpretation.

```r

library(tidyverse)
library(Rtsne)
library(patchwork)
library(cluster)


# 1) Load data 
data <- read.csv(gzfile("~/Desktop/codex_spleen2.csv.gz"), check.names = FALSE)
df <- data

# Fix id column 
if ("Unnamed: 0" %in% names(df)) df <- df %>% rename(id = `Unnamed: 0`)
if (names(df)[1] == "") names(df)[1] <- "id"

stopifnot(all(c("id", "x", "y") %in% names(df)))


# 2) Define columns
meta_cols <- intersect(c("id", "x", "y", "area"), names(df))
feat_cols <- setdiff(names(df), meta_cols)

# Make sure features are numeric
df[feat_cols] <- lapply(df[feat_cols], function(v) as.numeric(v))


# Total signal is common CODEX QC metric
df$total_signal <- rowSums(df[feat_cols], na.rm = TRUE)

# Robust cutoffs 
signal_low <- quantile(df$total_signal, 0.01, na.rm = TRUE)

df_qc <- df %>%
  filter(total_signal > signal_low) %>%
  drop_na(x, y)

# If area exists, filter tiny objects too
if ("area" %in% names(df_qc)) {
  area_low <- quantile(df_qc$area, 0.01, na.rm = TRUE)
  df_qc <- df_qc %>% filter(area > area_low)
}

message("QC kept ", nrow(df_qc), " / ", nrow(df), " cells")


# 4) Transform + scale (CODEX standard: arcsinh)

cofactor <- 5
X <- as.matrix(df_qc[feat_cols])
X_asinh <- asinh(X / cofactor)
X_scaled <- scale(X_asinh)

# 5) PCA and choose PCs (80% variance, cap 30)

pca <- prcomp(X_scaled, center = TRUE, scale. = FALSE)
var_expl <- (pca$sdev^2) / sum(pca$sdev^2)
cumvar <- cumsum(var_expl)
npcs <- which(cumvar >= 0.80)[1]
npcs <- min(npcs, 30)
pc_mat <- pca$x[, 1:npcs, drop = FALSE]


# 6) tSNE on PCs
set.seed(1)
n <- nrow(pc_mat)
perp <- min(30, max(5, floor((n - 1) / 3) - 1))  # safe perplexity
ts <- Rtsne(pc_mat, dims = 2, perplexity = perp, verbose = FALSE, max_iter = 800)
emb <- data.frame(tSNE1 = ts$Y[, 1], tSNE2 = ts$Y[, 2])


# 7) Choose k via silhouette and run kmeans
set.seed(1)
k_grid <- 3:10
d <- dist(pc_mat)

sil_mean <- sapply(k_grid, function(k) {
  km_tmp <- kmeans(pc_mat, centers = k, nstart = 20)
  mean(silhouette(km_tmp$cluster, d)[, "sil_width"])
})

k_best <- k_grid[which.max(sil_mean)]
message("Chosen k by silhouette: ", k_best)

set.seed(1)
km <- kmeans(pc_mat, centers = k_best, nstart = 50)
cluster <- factor(km$cluster)


# 8) Build plotting dataframe
plot_df <- df_qc %>%
  select(all_of(meta_cols), total_signal) %>%
  bind_cols(emb) %>%
  mutate(cluster = cluster)


# 9) Differential expression (Wilcoxon + median diff) for marker discovery
markers_tbl <- map_dfr(levels(cluster), function(cl) {
  in_cl <- cluster == cl
  out_cl <- cluster != cl
  
  pvals <- apply(X_asinh, 2, function(v) wilcox.test(v[in_cl], v[out_cl])$p.value)
  eff   <- apply(X_asinh, 2, function(v) median(v[in_cl]) - median(v[out_cl]))
  
  tibble(cluster = cl, marker = colnames(X_asinh), med_diff = eff, p_value = pvals) %>%
    arrange(desc(med_diff))
})

top_markers_tbl <- markers_tbl %>%
  group_by(cluster) %>%
  slice_head(n = 8) %>%
  ungroup()

print(top_markers_tbl)


# 10) Cell-type module scores (at least two cell types)
score_module <- function(mat, genes) {
  g <- intersect(genes, colnames(mat))
  if (length(g) == 0) return(rep(NA_real_, nrow(mat)))
  rowMeans(mat[, g, drop = FALSE])
}

# Common spleen/CODEX markers 
plot_df$score_T   <- score_module(X_asinh, c("CD3e", "CD3", "CD4", "CD8", "CD45RO", "CD45"))
plot_df$score_B   <- score_module(X_asinh, c("CD20", "CD21", "CD35", "CD45"))
plot_df$score_Mac <- score_module(X_asinh, c("CD68", "CD163", "HLA.DR"))
plot_df$score_End <- score_module(X_asinh, c("CD31", "CD34", "CollagenIV"))
plot_df$score_SMA <- score_module(X_asinh, c("SMActin"))

cl_means <- plot_df %>%
  group_by(cluster) %>%
  summarise(
    T = mean(score_T, na.rm = TRUE),
    B = mean(score_B, na.rm = TRUE),
    Mac = mean(score_Mac, na.rm = TRUE),
    End = mean(score_End, na.rm = TRUE),
    SMA = mean(score_SMA, na.rm = TRUE),
    .groups = "drop"
  )

print(cl_means)

cl_T <- cl_means$cluster[which.max(cl_means$T)]
cl_B <- cl_means$cluster[which.max(cl_means$B)]
message("Putative T-cell cluster: ", cl_T)
message("Putative B-cell cluster: ", cl_B)

plot_df$T_like <- ifelse(plot_df$cluster == cl_T, "T-like", "Other")
plot_df$B_like <- ifelse(plot_df$cluster == cl_B, "B-like", "Other")

p_T <- wilcox.test(plot_df$score_T[plot_df$cluster == cl_T], plot_df$score_T[plot_df$cluster != cl_T])$p.value
p_B <- wilcox.test(plot_df$score_B[plot_df$cluster == cl_B], plot_df$score_B[plot_df$cluster != cl_B])$p.value

fmt_p <- function(p) ifelse(is.na(p), "NA", ifelse(p < 2.2e-16, "<2.2e-16", signif(p, 3)))

# 11) VISUALIZATION 

# A) Downsample helper for spatial-only plots (keeps structure, reduces overplotting)
set.seed(1)
max_spatial_pts <- 60000
idx_spatial <- if (nrow(plot_df) > max_spatial_pts) sample(seq_len(nrow(plot_df)), max_spatial_pts) else seq_len(nrow(plot_df))
sp_df <- plot_df[idx_spatial, ]

pt_size  <- 0.12
pt_alpha <- 0.25

# Panel 1: tSNE colored by cluster (points can be a bit bigger)
p_tsne_cluster <- ggplot(plot_df, aes(tSNE1, tSNE2, color = cluster)) +
  geom_point(size = 0.35, alpha = 0.65) +
  theme_minimal() +
  labs(title = "tSNE (PCA space) colored by cluster")

# Panel 2: spatial map colored by cluster (FIX: small + transparent + fixed aspect)
p_spatial_cluster <- ggplot(sp_df, aes(x, y, color = cluster)) +
  geom_point(size = pt_size, alpha = pt_alpha) +
  coord_fixed() +
  scale_y_reverse() +
  theme_minimal() +
  labs(title = "Spatial map: k-means clusters",
       subtitle = paste0("k=", k_best, ", PCs=", npcs, ", tSNE perplexity=", perp))

# Panel 3: cluster centroid (median x,y per cluster) to see global organization
p_cluster_centroids <- plot_df %>%
  group_by(cluster) %>%
  summarise(x = median(x), y = median(y), n = n(), .groups = "drop") %>%
  ggplot(aes(x, y, color = cluster, size = n)) +
  geom_point(alpha = 0.9) +
  coord_fixed() +
  scale_y_reverse() +
  theme_minimal() +
  labs(title = "Cluster centroids (median position)", subtitle = "Point size = cluster cell count") +
  guides(size = "none")

# Panel 4: hex-binned cluster composition to reveal tissue regions (major fix)
# For each hex-bin, color by the most frequent cluster in that bin.
bin_w <- (max(plot_df$x) - min(plot_df$x)) / 40  # adjust for resolution
bin_h <- (max(plot_df$y) - min(plot_df$y)) / 40

hex_df <- plot_df %>%
  mutate(
    xb = floor((x - min(x)) / bin_w),
    yb = floor((y - min(y)) / bin_h)
  ) %>%
  group_by(xb, yb, cluster) %>%
  summarise(n = n(), .groups = "drop") %>%
  group_by(xb, yb) %>%
  slice_max(order_by = n, n = 1, with_ties = FALSE) %>%
  ungroup() %>%
  mutate(
    x_center = min(plot_df$x) + (xb + 0.5) * bin_w,
    y_center = min(plot_df$y) + (yb + 0.5) * bin_h
  )

p_hex_clusters <- ggplot(hex_df, aes(x_center, y_center, fill = cluster)) +
  geom_tile(width = bin_w, height = bin_h, alpha = 0.9) +
  coord_fixed() +
  scale_y_reverse() +
  theme_minimal() +
  labs(title = "Spatial structure (binned clusters)", subtitle = "Each tile shows dominant cluster in that region")

# Panel 5: Spatial T-like vs others (FIX: small + transparent; also uses downsample)
p_spatial_T <- ggplot(sp_df, aes(x, y, color = T_like)) +
  geom_point(size = pt_size, alpha = pt_alpha) +
  coord_fixed() +
  scale_y_reverse() +
  theme_minimal() +
  labs(title = "Spatial: putative T-cell region",
       subtitle = paste0("Cluster ", cl_T, " highest T-score; Wilcox p ", fmt_p(p_T))) +
  theme(legend.title = element_blank())

# Panel 6: Spatial B-like vs others
p_spatial_B <- ggplot(sp_df, aes(x, y, color = B_like)) +
  geom_point(size = pt_size, alpha = pt_alpha) +
  coord_fixed() +
  scale_y_reverse() +
  theme_minimal() +
  labs(title = "Spatial: putative B-cell region",
       subtitle = paste0("Cluster ", cl_B, " highest B-score; Wilcox p ", fmt_p(p_B))) +
  theme(legend.title = element_blank())

# Panel 7/8: Marker expression maps (auto-handle missing markers)
plot_marker_spatial <- function(marker_name, title_txt) {
  if (!(marker_name %in% colnames(X_asinh))) {
    return(ggplot() + theme_void() + labs(title = paste0(title_txt, "\n(Missing: ", marker_name, ")")))
  }
  tmp <- sp_df
  tmp$val <- X_asinh[idx_spatial, marker_name]
  ggplot(tmp, aes(x, y, color = val)) +
    geom_point(size = pt_size, alpha = pt_alpha) +
    coord_fixed() +
    scale_y_reverse() +
    theme_minimal() +
    labs(title = title_txt, color = "arcsinh")
}

p_CD3e <- plot_marker_spatial("CD3e", "CD3e (T-cell marker)")
p_CD20 <- plot_marker_spatial("CD20", "CD20 (B-cell marker)")

# 12) Final multi-panel figure (clear tissue structure)
final_plot <- (p_tsne_cluster | p_cluster_centroids) /
  (p_hex_clusters | p_spatial_cluster) /
  (p_spatial_T | p_spatial_B) /
  (p_CD3e | p_CD20)

final_plot

ggsave("CODEX_spleen_multi_panel_fixed.png", final_plot, width = 14, height = 14, dpi = 300)


# 13) quick tissue-structure hint table
structure_hint <- cl_means %>%
  mutate(
    hint = case_when(
      (T > quantile(T, 0.75, na.rm=TRUE) & B > quantile(B, 0.75, na.rm=TRUE)) ~ "White pulp-like (lymphoid)",
      (Mac > quantile(Mac, 0.75, na.rm=TRUE) & End > quantile(End, 0.75, na.rm=TRUE)) ~ "Red pulp-like (macrophage/vascular)",
      (End > quantile(End, 0.75, na.rm=TRUE) & SMA > quantile(SMA, 0.75, na.rm=TRUE)) ~ "Artery/Vein-like (vascular + SMA)",
      TRUE ~ "Mixed/unclear"
    )
  )

print(structure_hint)
write.csv(structure_hint, "CODEX_cluster_structure_hints.csv", row.names = FALSE)
```
