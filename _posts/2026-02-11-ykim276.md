### 1
This multi-panel figure identifies and characterizes a transcriptionally distinct population of cells (Cluster 6) in the Xenium kidney dataset. Raw gene counts were normalized using counts-per-million and log-transformed to make expression levels comparable across cells. Principal component analysis (PCA) was used to reduce dimensionality, followed by k-means clustering on the top principal components to define transcriptionally similar groups. In tSNE space (Panel A), Cluster 6 forms a compact and partially separated region, indicating that these cells share a coherent transcriptional identity. When mapped back to physical tissue space (Panel B), the same cluster shows structured spatial organization rather than random distribution, supporting that this is a biologically meaningful population rather than a clustering artifact.

Differential expression analysis comparing Cluster 6 to all other cells reveals strong enrichment of genes such as Bst1, Fyb, Cd53, Cd44, Lair1, Slc14a2, Runx1, and Socs3 (Panel C). The dot plot shows that these genes are not only statistically enriched but also expressed in a substantial fraction of Cluster 6 cells. Many of these markers are well known to be associated with immune or hematopoietic lineages. For example, Bst1 (CD157) and Cd53 are leukocyte-associated surface proteins, Lair1 functions as an immune inhibitory receptor, and Runx1 plays a key role in hematopoietic differentiation. The enrichment of multiple immune-related genes strongly suggests that Cluster 6 represents an immune cell population rather than epithelial kidney cells.

The marker gene Bst1 further supports this interpretation. In reduced space (Panel D), Bst1 expression is concentrated within the same region occupied by Cluster 6, confirming its specificity for this transcriptional population. In physical tissue space (Panel E), Bst1-positive cells are distributed throughout the tissue in a pattern consistent with immune cell presence rather than structured nephron segments. Together, the transcriptional distinctness of Cluster 6, the enrichment of canonical immune markers, and the spatial localization of Bst1 expression provide strong evidence that this cluster represents an immune cell population within the kidney.

library(ggplot2)
library(patchwork)
library(Rtsne)

set.seed(123)

# ------------------------------------------------------------
# 0) Read Xenium data
# ------------------------------------------------------------
data <- read.csv(
  "~/Documents/GitHub/genomic-data-visualization-2026/data/Xenium-IRI-ShamR_matrix.csv.gz",
  check.names = FALSE
)

# ------------------------------------------------------------
# 1) Build pos + expression (ROBUST to read.csv type issues)
# ------------------------------------------------------------
stopifnot(all(c("x", "y") %in% colnames(data)))

cell_id <- data[[1]]

pos <- data[, c("x", "y")]
rownames(pos) <- cell_id

# genes assumed to start at column 4
gexp_df <- data[, 4:ncol(data)]
colnames(gexp_df) <- make.unique(colnames(gexp_df))   # avoid duplicate gene-name issues

# convert to numeric matrix
gexp <- as.matrix(sapply(gexp_df, function(col) as.numeric(col)))
rownames(gexp) <- cell_id
gexp[is.na(gexp)] <- 0

# ------------------------------------------------------------
# 2) Normalize + log transform
# ------------------------------------------------------------
totgexp <- rowSums(gexp)
totgexp[totgexp == 0] <- 1
mat <- log10(gexp / totgexp * 1e6 + 1)

# ------------------------------------------------------------
# 3) Limit to highly-variable genes (speed + signal)
# ------------------------------------------------------------
n_hvg <- 300
gene_var <- apply(mat, 2, var)
gene_var[is.na(gene_var)] <- -Inf

hvg <- names(sort(gene_var, decreasing = TRUE))[1:min(n_hvg, ncol(mat))]
matsub <- mat[, hvg, drop = FALSE]

# ------------------------------------------------------------
# 4) PCA (optional scaling supported)
# ------------------------------------------------------------
use_scale <- FALSE  # set TRUE to try "optional scaling"
pcs <- prcomp(matsub, center = TRUE, scale. = use_scale)
top_pcs <- pcs$x[, 1:min(10, ncol(pcs$x)), drop = FALSE]

# ------------------------------------------------------------
# 5) Elbow plot for k (choose k_clusters)
# ------------------------------------------------------------
Ks <- 2:15
wss <- sapply(Ks, function(k) kmeans(top_pcs, centers = k, nstart = 20)$tot.withinss)
elbow_df <- data.frame(k = Ks, wss = wss)

p_elbow <- ggplot(elbow_df, aes(k, wss)) +
  geom_line() + geom_point() +
  theme_classic() +
  labs(title = "Elbow plot (k-means on top PCs)", x = "k", y = "Total withinss")
print(p_elbow)

k_clusters <- 9  # change after inspecting elbow plot

# ------------------------------------------------------------
# 6) k-means clustering (on PCs)
# ------------------------------------------------------------
km <- kmeans(top_pcs, centers = k_clusters, nstart = 25)
clusters <- factor(km$cluster)
names(clusters) <- rownames(top_pcs)

# master DF with correct rownames (cell IDs)
df <- data.frame(
  x = pos$x,
  y = pos$y,
  PC1 = pcs$x[, 1],
  PC2 = pcs$x[, 2],
  cluster = clusters,
  row.names = rownames(pos)
)

# overview (optional but helpful to choose your cluster)
p_all_pca <- ggplot(df, aes(PC1, PC2, color = cluster)) +
  geom_point(size = 0.35, alpha = 0.6) +
  theme_classic() +
  labs(title = "All clusters in PCA space", color = "Cluster")
print(p_all_pca)

p_all_space <- ggplot(df, aes(x, y, color = cluster)) +
  geom_point(size = 0.35, alpha = 0.6) +
  coord_fixed() +
  theme_void() +
  labs(title = "All clusters in physical space", color = "Cluster")
print(p_all_space)

# ------------------------------------------------------------
# 7) tSNE (reduced-space visualization)
# ------------------------------------------------------------
tsne <- Rtsne(top_pcs, dims = 2, perplexity = 30)
df$tSNE1 <- tsne$Y[, 1]
df$tSNE2 <- tsne$Y[, 2]

p_all_tsne <- ggplot(df, aes(tSNE1, tSNE2, color = cluster)) +
  geom_point(size = 0.35, alpha = 0.6) +
  theme_classic() +
  labs(title = "All clusters in tSNE space", color = "Cluster")
print(p_all_tsne)

# ------------------------------------------------------------
# 8) Pick ONE cluster of interest
# ------------------------------------------------------------
cluster_of_interest <- "6"  # CHANGE after inspecting plots

df$highlight <- ifelse(df$cluster == cluster_of_interest,
                       paste0("Cluster ", cluster_of_interest),
                       "Other")

# IMPORTANT FIX: build named color map first (prevents "unexpected '='")
cluster_label <- paste0("Cluster ", cluster_of_interest)
col_map <- c("grey85", "royalblue")
names(col_map) <- c("Other", cluster_label)

# ------------------------------------------------------------
# 9) Differential expression (cluster vs all others)
# ------------------------------------------------------------
in_cells  <- rownames(df)[df$cluster == cluster_of_interest]
out_cells <- rownames(df)[df$cluster != cluster_of_interest]

mat_in  <- mat[in_cells, , drop = FALSE]
mat_out <- mat[out_cells, , drop = FALSE]

mean_in  <- colMeans(mat_in)
mean_out <- colMeans(mat_out)

diff_log10CPM <- mean_in - mean_out
pct_in  <- colMeans(mat_in  > 0)
pct_out <- colMeans(mat_out > 0)
specificity <- pct_in - pct_out

pval <- sapply(colnames(mat), function(g) {
  wilcox.test(mat_in[, g], mat_out[, g], alternative = "greater")$p.value
})
padj <- p.adjust(pval, method = "BH")

de <- data.frame(
  gene = colnames(mat),
  diff_log10CPM = diff_log10CPM[colnames(mat)],
  pct_in = pct_in[colnames(mat)],
  pct_out = pct_out[colnames(mat)],
  specificity = specificity[colnames(mat)],
  pval = pval,
  padj = padj
)

de <- de[order(-de$diff_log10CPM, de$padj), ]
write.csv(de, "Xenium_DE_table_cluster_interest.csv", row.names = FALSE)

# marker gene (auto; you can override to a known marker)
marker_gene <- de$gene[1]

# ------------------------------------------------------------
# 10) REQUIRED 5-PANEL FIGURE
#   1) cluster in reduced space (tSNE)
#   2) cluster in physical space
#   3) DE genes panel
#   4) one DE gene in reduced space
#   5) one DE gene in physical space
# ------------------------------------------------------------

# (1) Cluster of interest in reduced space
p1 <- ggplot(df, aes(tSNE1, tSNE2, color = highlight)) +
  geom_point(size = 0.35, alpha = 0.55) +
  theme_classic() +
  labs(title = paste0("Cluster ", cluster_of_interest, " in tSNE space"), color = "") +
  scale_color_manual(values = col_map)

# (2) Cluster of interest in physical space
p2 <- ggplot(df, aes(x, y, color = highlight)) +
  geom_point(size = 0.35, alpha = 0.55) +
  coord_fixed() +
  theme_void() +
  labs(title = paste0("Cluster ", cluster_of_interest, " in physical space"), color = "") +
  scale_color_manual(values = col_map)

# (3) DE genes panel (dot plot of top 10 by padj; fallback if too few)
de_sig <- de[de$padj < 0.05, ]
if (nrow(de_sig) < 10) de_sig <- de
top10 <- head(de_sig, 10)

dot_df <- do.call(rbind, lapply(top10$gene, function(g) {
  data.frame(
    gene = g,
    group = c("Cluster", "Other"),
    mean_expr = c(mean(mat_in[, g]), mean(mat_out[, g])),
    pct_expr  = c(mean(mat_in[, g] > 0), mean(mat_out[, g] > 0))
  )
}))
dot_df$gene <- factor(dot_df$gene, levels = rev(top10$gene))

p3 <- ggplot(dot_df, aes(group, gene)) +
  geom_point(aes(size = pct_expr, color = mean_expr)) +
  theme_classic() +
  labs(
    title = paste0("Top DE genes (Cluster ", cluster_of_interest, " vs Other)"),
    size  = "% expressing",
    color = "Mean log10(CPM+1)"
  ) +
  scale_size(range = c(2, 8))

# Add marker gene expression
df$marker_expr <- mat[rownames(df), marker_gene]

# (4) Marker gene in reduced space
p4 <- ggplot(df, aes(tSNE1, tSNE2, color = marker_expr)) +
  geom_point(size = 0.35, alpha = 0.65) +
  theme_classic() +
  labs(title = paste0(marker_gene, " expression in tSNE space"),
       color = "log10(CPM+1)")

# (5) Marker gene in physical space
p5 <- ggplot(df, aes(x, y, color = marker_expr)) +
  geom_point(size = 0.35, alpha = 0.65) +
  coord_fixed() +
  theme_void() +
  labs(title = paste0(marker_gene, " expression in physical space"),
       color = "log10(CPM+1)")

final_plot <- (p1 | p2) / (p3) / (p4 | p5) +
  plot_annotation(title = paste0("Xenium: Cluster ", cluster_of_interest,
                                 " | marker: ", marker_gene))

print(final_plot)
ggsave("Xenium_HW_multiplot.png", final_plot, width = 14, height = 16, dpi = 300)

# ------------------------------------------------------------
# 11) Summary for write-up
# ------------------------------------------------------------
cat("\n===== SUMMARY =====\n")
cat("Scaling used in PCA:", use_scale, "\n")
cat("k (k-means centers):", k_clusters, "\n")
cat("Cluster of interest:", cluster_of_interest, "\n")
cat("Marker gene chosen:", marker_gene, "\n\n")
cat("Top 10 DE genes:\n")
print(head(de[, c("gene", "diff_log10CPM", "specificity", "padj")], 10))
