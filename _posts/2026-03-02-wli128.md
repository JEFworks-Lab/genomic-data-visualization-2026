# 1. Description
In this data visualization, I compared standard kmeans clustering with STdeconvolve's deconvolved latent topics. In the top row of panels, I used kmeans clustering and in the bottom row, I used STdeconvolve latent topics. Then, in the left column I show these results in xy space. Then the middle column, the same clustering in tsne space. Finally, using these clusters I plotted a dotplot of the top 5 differentially expressed genes in each cluster/topic. 

I chose to investigate the collecting duct cells because they are the most distinct cell types in the dataset. All of the collecting duct cells are expected to localize in the center of the medulla of the kidney in xy space. Additionally, in tsne space, the collecting duct cells cluster together and very far away from the other cell types. 

In the standard kmeans clustering we see that with high confidence the collecting duct cells localize in the region we expect in the center of the medulla. Furthermore the STdeconvolved result also shows that the same region shows that it is nearly complete completely made up of collecting duct cells. Thus, I expected the top genes of this region to be very similar between both STdeconvolve and normal kmeans clustering. However looking at the top 5 most significantly expressed genes in this group, there was no overlap at all. I believe this shows how the two methods differ in defining and weighting gene importance. Therefore, even though cells in these categories heavily overlap, the most definitive genes of each group could be different.

```r
library(STdeconvolve)
library(scatterbar)
library(ggplot2)
library(Rtsne)
library(dplyr)
library(patchwork)

data <- read.csv('/Users/willli/Documents/BME 25-26/Genomic Data Visualization/genomic-data-visualization-2026/data/Visium-IRI-ShamR_matrix.csv.gz')
rownames(data) <- data[, 1]
pos <- data[, c('x', 'y')]
gexp <- data[, 4:ncol(data)]

genes_xen <- read.csv('/Users/willli/Documents/BME 25-26/Genomic Data Visualization/genomic-data-visualization-2026/data/Xenium-IRI-ShamR_matrix.csv.gz')
genes_xen <- colnames(genes_xen)[4:ncol(genes_xen)]
genes.have <- intersect(genes_xen, colnames(gexp))
gexp.sub <- gexp[, genes.have]   

totgexp <- rowSums(gexp)
mat <- log10(gexp / totgexp * 1e6 + 1)
rownames(mat) <- data$X

# Dimenstionality reduction
pcs <- prcomp(mat, center = TRUE, scale = FALSE)
toppcs <- pcs$x[, 1:20]
tsne <- Rtsne::Rtsne(toppcs, dims = 2, perplexity = 30)
emb <- tsne$Y
rownames(emb) <- rownames(mat)
colnames(emb) <- c('tsne1', 'tsne2')

# K-Means Clusters -------------------------------------------------
set.seed(42)
clusters <- as.factor(kmeans(toppcs, centers = 7)$cluster)
names(clusters) <- rownames(mat)
levels(clusters)[levels(clusters) == "5"] <- "Collecting duct"

df_km <- data.frame(pos, emb, clusters)

p1_xy <- ggplot(df_km, aes(x = x, y = y, col = clusters)) +
  geom_point(size = 0.8) +
  theme_bw() +
  labs(title = 'K-means clusters\n(physical space)', color = 'Cluster') +
  guides(color = guide_legend(override.aes = list(size = 3)))


p1_tsne <- ggplot(df_km, aes(x = tsne1, y = tsne2, col = clusters)) +
  geom_point(size = 0.8) +
  theme_bw() +
  labs(title = 'K-means clusters\n(tSNE space)', color = 'Cluster') +
  guides(color = guide_legend(override.aes = list(size = 3)))

vg <- apply(mat, 2, var)
candidate_genes <- names(sort(vg, decreasing = TRUE))[1:500]

sig_out <- data.frame(matrix(nrow = length(candidate_genes), ncol = 0))

for (cl in sort(unique(clusters))) {
  idx1 <- which(clusters == cl)
  idx2 <- which(clusters != cl)
  
  sig_out[[as.character(cl)]] <- sapply(candidate_genes, function(gene) {
    x1 <- mat[idx1, gene]
    x2 <- mat[idx2, gene]
    res <- wilcox.test(x1, x2, alternative = 'greater')$p.value
    -log10(res)
  })
}
rownames(sig_out) <- candidate_genes

top5_km <- do.call(rbind, lapply(colnames(sig_out), function(cl) {
  vals <- sig_out[[cl]]
  names(vals) <- rownames(sig_out)
  top <- sort(vals, decreasing = TRUE)[1:5]
  data.frame(
    cluster = cl,
    gene = names(top),
    sig = as.numeric(top),
    stringsAsFactors = FALSE
  )
}))

top5_genes_unique <- unique(top5_km$gene)
avg_km <- sapply(top5_genes_unique, function(g) tapply(mat[, g], clusters, mean, na.rm = TRUE))
avg_km_long <- as.data.frame(as.table(avg_km))
colnames(avg_km_long) <- c("cluster", "gene", "avg_expr")

df_dot_km <- merge(avg_km_long,
                   top5_km[, c("cluster", "gene", "sig")],
                   by = c("cluster", "gene"), all.x = TRUE)
df_dot_km$sig[is.na(df_dot_km$sig)] <- 0
df_dot_km$sig[is.infinite(df_dot_km$sig)] <- max(df_dot_km$sig[is.finite(df_dot_km$sig)])

df_dot_km$gene <- factor(df_dot_km$gene, levels = unique(top5_km$gene))
df_dot_km$cluster <- factor(df_dot_km$cluster)

p1_dot <- ggplot(df_dot_km, aes(x = cluster, y = gene, size = sig, color = avg_expr)) +
  geom_point(aes(size = ifelse(sig >= 3, sig, NA), color = avg_expr)) +
  theme_bw() +
  theme(axis.text.y = element_text(size = 7),
        axis.text.x = element_text(angle = 90, hjust = 1, vjust = 0.5)) +
  labs(title = 'Top DEGs per\nK-means cluster',
       size = '-log10(p-val)',
       color = 'log10 Avg expr',
       x = 'Cluster', y = 'Gene')

# STdeconvolve ----------------------------------
set.seed(42)
ldas <- fitLDA(gexp.sub, Ks = 7)
optLDA <- optimalModel(models = ldas, opt = 'min')
results <- getBetaTheta(optLDA, perc.filt = 0.05, betaScale = 1000)

deconProp <- results$theta
deconGexp <- results$beta

colnames(deconProp)[3] <- "Collecting duct"

p2_xy <- scatterbar(deconProp, pos,
                    size_x = 90, size_y = 90,
                    padding_x = 1, padding_y = 1) +
  theme_bw() +
  labs(title = 'STdeconvolve topics\n(physical space)', y ='y')

pos_tsne <- data.frame(x = emb[, 1], y = emb[, 2])
rownames(pos_tsne) <- rownames(emb)

common_spots <- intersect(rownames(deconProp), rownames(pos_tsne))
deconProp_ord <- deconProp[common_spots, ]
pos_tsne_ord <- pos_tsne[common_spots, ]

tsne_xrange <- diff(range(pos_tsne_ord$x))
tsne_yrange <- diff(range(pos_tsne_ord$y))
tsne_sx <- tsne_xrange / sqrt(nrow(pos_tsne_ord)) * 1.5
tsne_sy <- tsne_yrange / sqrt(nrow(pos_tsne_ord)) * 1.5

p2_tsne <- scatterbar(deconProp_ord, pos_tsne_ord,
                      size_x = tsne_sx, size_y = tsne_sy,
                      padding_x = 0.1, padding_y = 0.1) +
  theme_bw() +
  labs(title = 'STdeconvolve topics\n(tSNE space)',
       x = 'tsne1', y = 'tsne2', color = 'topic')

dominant_topic <- as.factor(apply(deconProp, 1, which.max))
names(dominant_topic) <- rownames(deconProp)

common_spots_mat <- intersect(names(dominant_topic), rownames(mat))
dominant_topic_sub <- dominant_topic[common_spots_mat]

candidate_genes_st <- intersect(colnames(mat), genes.have)

vg_st <- apply(mat[common_spots_mat, candidate_genes_st], 2, var)
candidate_genes_st <- names(sort(vg_st, decreasing = TRUE))[1:min(500, length(vg_st))]

sig_out_st <- data.frame(matrix(nrow = length(candidate_genes_st), ncol = 0))

for (cl in sort(unique(dominant_topic_sub))) {
  idx1 <- which(dominant_topic_sub == cl)
  idx2 <- which(dominant_topic_sub != cl)
  
  sig_out_st[[as.character(cl)]] <- sapply(candidate_genes_st, function(gene) {
    x1 <- mat[common_spots_mat[idx1], gene]
    x2 <- mat[common_spots_mat[idx2], gene]
    res <- wilcox.test(x1, x2, alternative = 'greater')$p.value
    -log10(res)
  })
}
rownames(sig_out_st) <- candidate_genes_st

top5_st <- do.call(rbind, lapply(colnames(sig_out_st), function(cl) {
  vals <- sig_out_st[[cl]]
  names(vals) <- rownames(sig_out_st)
  top <- sort(vals, decreasing = TRUE)[1:5]
  data.frame(
    topic = paste0("K", cl),
    gene = names(top),
    sig = as.numeric(top),
    stringsAsFactors = FALSE
  )
}))

top5_genes_st_unique <- unique(top5_st$gene)
avg_st <- sapply(top5_genes_st_unique, function(g) {
  tapply(mat[common_spots_mat, g], dominant_topic_sub, mean, na.rm = TRUE)
})
avg_st_long <- as.data.frame(as.table(avg_st))
colnames(avg_st_long) <- c("topic", "gene", "avg_expr")
avg_st_long$topic <- paste0("K", avg_st_long$topic)

df_dot_st <- merge(avg_st_long,
                   top5_st[, c("topic", "gene", "sig")],
                   by = c("topic", "gene"), all.x = TRUE)
df_dot_st$sig[is.na(df_dot_st$sig)] <- 0
df_dot_st$sig[is.infinite(df_dot_st$sig)] <- max(df_dot_st$sig[is.finite(df_dot_st$sig)])

df_dot_st$gene <- factor(df_dot_st$gene, levels = unique(top5_st$gene))
df_dot_st$topic <- factor(df_dot_st$topic)
levels(df_dot_st$topic)[levels(df_dot_st$topic) == "K3"] <- "Collecting duct"
p2_dot <- ggplot(df_dot_st, aes(x = topic, y = gene, size = sig, color = avg_expr)) +
  geom_point(aes(size = ifelse(sig >= 3, sig, NA), color = avg_expr )) +
  theme_bw() +
  theme(axis.text.y = element_text(size = 7),
        axis.text.x = element_text(angle = 90, hjust = 1, vjust = 0.5)) +
  labs(title = 'Top DEGs per\nSTdeconvolve topic',
       size = '-log10(p-val)',
       color = 'log10 Avg expr',
       x = 'Topic', y = 'Gene')

# Final plot ---------
final_plot <- (p1_xy | p1_tsne | p1_dot) /
              (p2_xy | p2_tsne | p2_dot) +
  plot_annotation(
    title = 'K-means vs STdeconvolve Analysis in Visium Mouse Kidney Data',
    theme = theme(plot.title = element_text(size = 16, face = 'bold', hjust = 0.5))
  )

final_plot
```
