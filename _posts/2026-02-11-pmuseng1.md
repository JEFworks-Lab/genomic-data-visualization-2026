### Description
Cluster 3 forms a clearly separated group in PCA space, indicating that it represents a transcriptionally distinct cell population. Differential expression analysis (Wilcoxon rank-sum test, one-sided) identified Aqp1 among the top marker genes enriched in this cluster. 
Aqp1 encodes Aquaporin-1, a water channel protein that is highly expressed in kidney proximal tubule epithelial cells and plays a critical role in renal water reabsorption (Nielsen et al., 1993; Rojek et al., 2006). In PCA space, Aqp1 expression strongly overlaps with cluster 3, supporting its transcriptional identity.
Although the spatial expression pattern of Aqp1 does not show sharply localized enrichment confined to a narrow anatomical structure, the strong transcriptional separation and marker enrichment together suggest that cluster 3 likely corresponds to a proximal tubule–enriched compartment or a related epithelial population. 
The more diffuse spatial pattern may reflect biological heterogeneity within the tissue section, partial mixing of adjacent nephron segments, or technical smoothing inherent to spatial transcriptomics measurements.
REFERENCE: Marín-Vásquez, J. M., et al. (2020). Spatial transcriptomics of the adult mammalian kidney reveals proximal tubule zonation and segment-specific gene signatures. Kidney International, 97(4), 712–723.

library(ggplot2)
library(patchwork)   # for multi-panel figures
library(dplyr)

set.seed(42)

data <- read.csv("/Users/pmusenge/Desktop/genomic_data_vis/genomic-data-visualization-2026/data/Xenium-IRI-ShamR_matrix.csv.gz")
print(dim(data))


n_keep <- 10000
if (nrow(data) > n_keep) {
  data <- data[sample(1:nrow(data), n_keep), ]
}
print(dim(data))


pos <- data[, c("x", "y")]
rownames(pos) <- data[, 1]

gexp <- data[, 4:ncol(data)]
rownames(gexp) <- data[, 1]


print(head(pos))
print(gexp[1:5, 1:5])
print(dim(gexp))

# normalize
totgexp <- rowSums(gexp)
mat <- log10(gexp / totgexp * 1e6 + 1)

# PCA
pcs <- prcomp(mat)
df_base <- data.frame(pcs$x, pos)  
ggplot(df_base, aes(x=x, y=y, col=PC1)) + geom_point(cex=0.5)



# Kmeans
k <- 3                
pcs_for_kmeans <- 1:2  
km <- kmeans(pcs$x[, pcs_for_kmeans], centers = k)

clusters <- as.factor(km$cluster)
names(clusters) <- rownames(pcs$x)

df <- data.frame(df_base, clusters = clusters)
ggplot(df, aes(x=PC1, y=PC2, col=clusters)) + geom_point(cex=0.5) 
ggplot(df, aes(x=x, y=y, col=clusters)) + geom_point(cex=0.5)


# picking cluster of interest
cluster_interest <- 3   

cluster_cells <- names(clusters)[clusters == cluster_interest]
other_cells   <- names(clusters)[clusters != cluster_interest]


# Differential expression: Wilcoxon for all genes 
p_wilcox <- sapply(colnames(mat), function(gene) {
  x1 <- mat[cluster_cells, gene]
  x2 <- mat[other_cells, gene]
  wilcox.test(x1, x2, alternative = "greater")$p.value
})


eff_median <- sapply(colnames(mat), function(gene) {
  median(mat[cluster_cells, gene]) - median(mat[other_cells, gene])
})

de <- data.frame(
  gene = colnames(mat),
  p = p_wilcox,
  neglog10p = -log10(p_wilcox),
  med_diff = eff_median
) %>%
  arrange(p)

print(head(de, 30))
de_sorted <- de[order(de$p), ]
head(de_sorted, 20)

de_med_diff_sorted <- de[order(de$med_diff), ]
head(de_med_diff_sorted, 20)


# choosing a marker gene (ideally: small p AND positive med_diff)
marker_gene <- "Aqp1" 

stopifnot(marker_gene %in% colnames(mat)) #check

# Building the required 5 panels

# cluster in reduced space
p1 <- ggplot(df, aes(x = PC1, y = PC2, col = clusters)) +
  geom_point(size = 0.6, alpha = 0.6) +
  ggtitle("Clusters in PCA space (PC1 vs PC2)")

# cluster in physical space
p2 <- ggplot(df, aes(x = x, y = y, col = clusters)) +
  geom_point(size = 0.6, alpha = 0.6) +
  coord_fixed() +
  ggtitle("Clusters in tissue space")

# DE genes panel (top genes)
top_n <- 15
de_top <- de %>% slice(1:top_n) %>% mutate(gene = factor(gene, levels = rev(gene)))

p3 <- ggplot(de_top, aes(x = gene, y = neglog10p)) +
  geom_col() +
  coord_flip() +
  ggtitle(paste0("Cluster ", cluster_interest, " DE genes"))


# adding marker gene expression into df for plotting
df_gene <- data.frame(df, gene_expr = mat[rownames(df), marker_gene])

# marker gene in reduced space
p4 <- ggplot(df_gene, aes(x = PC1, y = PC2, col = gene_expr)) +
  geom_point(size = 0.6, alpha = 0.6) +
  ggtitle(paste0(marker_gene, " in PCA space"))

# marker gene in physical space
p5 <- ggplot(df_gene, aes(x = x, y = y, col = gene_expr)) +
  geom_point(size = 0.6, alpha = 0.6) +
  coord_fixed() +
  ggtitle(paste0(marker_gene, " in tissue space"))


final_fig <- (p1 | p2) / (p3 | p4) / p5 +
  plot_layout(heights = c(1, 1, 1.2)) &
  theme(
    plot.title = element_text(size = 11, face = "bold"),
    plot.margin = margin(20, 25, 15, 25) #AI HELP; Prompt: The image produced using the R code below has cluttered figure headings. How can I create more space between the figures for a cleaner image 
  )
final_fig
