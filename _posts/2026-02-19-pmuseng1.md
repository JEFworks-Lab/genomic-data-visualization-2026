### Description

Using the CODEX spleen dataset, I identified transcriptionally distinct immune cell populations by performing PCA on normalized protein expression, selecting the first 10 PCs based on the scree plot elbow, and choosing 
k=6 clusters from the WSS elbow curve, which reflects diminishing improvement beyond six groups. Cluster 4 showed the highest mean CD20 expression across clusters, and its top differentially expressed markers included CD20,
CD21, CD35, CD44, and HLA-DR—canonical B-cell and follicular markers—indicating this cluster represents a B-cell-rich white pulp region of the spleen. 
Although CD3e and CD4 were not among the top-15 DE markers for cluster 4, they appear strongly expressed in other clusters (e.g., cluster 1), consistent with T-cell-rich periarteriolar lymphoid sheath regions, and together 
the marker expression panels demonstrate complementary B-cell versus T-cell localization expected in splenic makeup. This pattern is biologically well supported, as CD20 and CD21 are established B-cell lineage markers involved 
in antigen presentation and B-cell activation, while CD3e and CD4 identify T-cell subsets in splenic immune compartments (Tedder et al., 2008; Mebius & Kraal, 2005). Overall, the clustering, DE analysis, and marker expression plots 
consistently identify a biologically meaningful B-cell follicle population in the CODEX spleen tissue.

```r

library(ggplot2)
library(patchwork)
library(dplyr)

set.seed(42)


data <- read.csv("/Users/pmusenge/Desktop/genomic_data_vis/genomic-data-visualization-2026/data/codex_spleen2.csv.gz")
print(dim(data))
print(dim(data))
head(data)
print(colnames(data)[1:10])

# Coordinates
# Coordinates + expression matrix (protein markers)
# Columns: 1=ID, 2=x, 3=y, 4=area, 5:...=markers
pos <- data[, c("x","y")]
rownames(pos) <- data[,1]

# Protein expression
expr <- data[,5:ncol(data)]   
rownames(expr) <- data[,1]

# Normalize
tot <- rowSums(expr)
mat <- log10(expr/tot*1e6 + 1)

# PCA
pcs <- prcomp(mat)
df_base <- data.frame(pcs$x, pos)

# Scree plot data
var_exp <- pcs$sdev^2 / sum(pcs$sdev^2)
df_scree <- data.frame(PC = 1:length(var_exp), var = var_exp)

p_scree <- ggplot(df_scree[1:25, ], aes(x = PC, y = var)) +
  geom_point() + geom_line() +
  ggtitle("Scree plot (variance explained)") +
  ylab("Variance explained") + xlab("PC")


# Choosing # PCs for clustering (based on your scree plot above)
n_pcs <- 10

# Elbow plot for choosing k (WSS)
k_grid <- 2:10
wss <- sapply(k_grid, function(k) {
  kmeans(pcs$x[, 1:n_pcs], centers = k)$tot.withinss
})
df_wss <- data.frame(k = k_grid, wss = wss)

p_wss <- ggplot(df_wss, aes(x = k, y = wss)) +
  geom_point() + geom_line() +
  ggtitle("Elbow plot (WSS vs k)") +
  xlab("k (number of clusters)") + ylab("Total within-cluster SS")


# Choosing k based on elbow plot and #of PCs above
k <- 6
km <- kmeans(pcs$x[, 1:n_pcs], centers = k)

clusters <- as.factor(km$cluster)
names(clusters) <- rownames(pcs$x)     # NOTE: IMPORTANT

df <- data.frame(df_base, clusters = clusters)


# Cluster plots
p1 <- ggplot(df, aes(x = PC1, y = PC2, col = clusters)) +
  geom_point(size = 0.7, alpha = 0.8) +
  ggtitle("Clusters in PCA space (PC1 vs PC2)")

p2 <- ggplot(df, aes(x = x, y = y, col = clusters)) +
  geom_point(size = 0.8, alpha = 0.85) +
  coord_fixed() +
  ggtitle("Clusters in tissue space")

#p1
#p2
# Cluster-wise marker summaries (to justify why I picked these clusters)
# Calculating mean marker expression by cluster (using normalized mat)
#AI Help. Prompt: help me with a function to calculate the average expression of a given marker gene in each cluster given these markers I have identified
cluster_mean <- function(marker_name) {
  if (!marker_name %in% colnames(mat)) return(rep(NA, length(levels(clusters))))
  sapply(levels(clusters), function(cl) {
    mean(mat[names(clusters)[clusters == cl], marker_name])
  })
}

mean_CD20 <- cluster_mean("CD20")
mean_CD3e <- cluster_mean("CD3e")

df_marker_means <- data.frame(
  cluster = factor(levels(clusters), levels = levels(clusters)),
  CD20 = mean_CD20,
  CD3e = mean_CD3e
)

df_marker_means_long <- df_marker_means %>%
  tidyr::pivot_longer(cols = c("CD20", "CD3e"), names_to = "marker", values_to = "mean_expr")

p_means <- ggplot(df_marker_means_long, aes(x = cluster, y = mean_expr, fill = marker)) +
  geom_col(position = "dodge") +
  ggtitle("Mean marker expression by cluster") +
  ylab("Mean normalized expression") + xlab("Cluster")


# Picking cluster of interest (based on mean marker calculations above)
# B-cell rich: highest mean CD20 == cluster 4
cluster_interest <- as.integer(as.character(df_marker_means$cluster[which.max(df_marker_means$CD20)]))
cat("Selected B-cell cluster (max CD20) =", cluster_interest, "\n")

# Also identifying T-cell rich cluster (max CD3e)
tcell_cluster <- as.integer(as.character(df_marker_means$cluster[which.max(df_marker_means$CD3e)]))
cat("Most T-cell-like cluster (max CD3e) =", tcell_cluster, "\n")


#  Differential expression (cluster_interest vs rest)
cells  <- names(clusters)[clusters == cluster_interest]
others <- names(clusters)[clusters != cluster_interest]

pvals <- sapply(colnames(mat), function(marker) {
  wilcox.test(mat[cells, marker], mat[others, marker], alternative = "greater")$p.value
})

de <- data.frame(
  marker = colnames(mat),
  p = pvals,
  neglog10 = -log10(pvals)
) %>% arrange(p)

print(head(de, 20))


# Handling p-values with -> Inf
de$neglog10 <- ifelse(is.infinite(de$neglog10),
                      max(de$neglog10[is.finite(de$neglog10)], na.rm = TRUE) + 10,
                      de$neglog10)


# DE bar plot
top_n <- 15
de_top <- de %>%
  slice(1:top_n) %>%
  mutate(marker = factor(marker, levels = rev(marker)))

p3 <- ggplot(de_top, aes(x = marker, y = neglog10)) +
  geom_col() +
  coord_flip() +
  ggtitle(paste0("Top DE markers (cluster ", cluster_interest, " vs rest)")) +
  xlab("Marker") + ylab("-log10(p)")

#p3

# Marker expression panels (PCA + tissue)
# Choosing markers that support white pulp claim strongly:  B-cell: CD20, CD21, CD35 and T-cell: CD3e, CD4
markers_to_plot <- c("CD20", "CD21", "CD35", "CD3e", "CD4")
markers_to_plot <- markers_to_plot[markers_to_plot %in% colnames(mat)]

df_mark <- df %>%
  mutate(id = rownames(df)) %>%
  select(id, x, y, PC1, PC2, clusters)

marker_long <- do.call(rbind, lapply(markers_to_plot, function(m) {
  data.frame(
    id = rownames(mat),
    marker = m,
    expr = mat[, m]
  )
}))

df_gene <- left_join(df_mark, marker_long, by = "id")

p4 <- ggplot(df_gene, aes(x = PC1, y = PC2, col = expr)) +
  geom_point(size = 0.7, alpha = 0.8) +
  facet_wrap(~ marker, nrow = 1) +
  ggtitle("Marker expression in PCA space") +
  labs(col = "expr")

#p4

p5 <- ggplot(df_gene, aes(x = x, y = y, col = expr)) +
  geom_point(size = 0.8, alpha = 0.85) +
  coord_fixed() +
  facet_wrap(~ marker, nrow = 1) +
  ggtitle("Marker expression in tissue space") +
  labs(col = "expr")

#p5

# Final multi-panel figure
final_fig <- (p_scree | p_wss) /
  (p1 | p2) /
  (p_means | p3) /
  p4 /
  p5 +
  plot_layout(heights = c(0.9, 1, 1.05, 1.1, 1.15)) +
  plot_annotation(
    title = "CODEX Spleen: Evidence for White Pulp (B-cell follicle + T-cell zone)",
    subtitle = paste0(
      "PCs chosen from scree plot (n_pcs=", n_pcs, "); k chosen from WSS elbow (k=", k,
      "). Cluster ", cluster_interest, " selected as B-cell–rich (max CD20); cluster ",
      tcell_cluster, " shows strongest CD3e (T-cell–rich)."
    ),
    theme = theme(
      plot.title = element_text(size = 16, face = "bold", hjust = 0.5),
      plot.subtitle = element_text(size = 11, hjust = 0.5)
    )
  ) &
  theme(
    plot.title = element_text(size = 11, face = "bold"),
    plot.margin = margin(20, 25, 15, 25)
  )

final_fig

# saving image
ggsave(
  "/Users/pmusenge/Desktop/genomic_data_vis/homework_scripts/HW5/hw5_codex_multi_panel.png",
  final_fig,
  width = 14,
  height = 14,
  dpi = 300
)
```
