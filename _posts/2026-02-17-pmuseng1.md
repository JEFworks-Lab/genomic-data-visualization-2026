### Description
In HW3 (Xenium), I clustered cells using k-means with k = 3 on the first two principal components. For HW4 (Visium), I kept the same workflow but re-evaluated both the number of PCs and clusters because Visium data are spot-based and contain mixtures of cell types. 
A scree plot of variance explained (see code below) showed that the first several PCs captured the major transcriptional structure before flattening, so I clustered using the first 10 PCs to retain biological signal while reducing noise. I then evaluated k from 2–10 using total within-cluster sum of squares (see code below) and observed an elbow around k ≈ 5, indicating 
diminishing improvement beyond five clusters. I therefore used k = 5, which also produced spatially coherent clusters in tissue space. Compared to Xenium single-cell data, Visium spots aggregate multiple neighboring cells, so fewer distinct transcriptional clusters are expected and marker gene signals appear more diffuse. 
Using these parameters, I identified a cluster enriched for proximal tubule markers (Aqp1, Slc34a1, Cyp2e1), consistent with the same cell-type I identified in the Xenium dataset.
REFERENCE: Marín-Vásquez, J. M., et al. (2020). Spatial transcriptomics of the adult mammalian kidney reveals proximal tubule zonation and segment-specific gene signatures. Kidney International, 97(4), 712–723.

```r
library(ggplot2)
library(patchwork)
library(dplyr)

set.seed(42)

data <- read.csv("/Users/pmusenge/Desktop/genomic_data_vis/genomic-data-visualization-2026/data/Visium-IRI-ShamR_matrix.csv.gz")
print(dim(data))
print(colnames(data)[1:6])


pos <- data[, c("x", "y")]
rownames(pos) <- data[, 1]  

gexp <- data[, 4:ncol(data)]         # genes start at column 4
rownames(gexp) <- data[, 1]


print(head(pos))
print(dim(gexp))

# Normalize
totgexp <- rowSums(gexp)
mat <- log10(gexp / totgexp * 1e6 + 1)


# PCA
pcs <- prcomp(mat)
df_base <- data.frame(pcs$x, pos)


# Kmeans
k <- 5              
pcs_for_kmeans <- 1:10    
km <- kmeans(pcs$x[, pcs_for_kmeans], centers = k)

clusters <- as.factor(km$cluster)
names(clusters) <- rownames(pcs$x)

df <- data.frame(df_base, clusters = clusters)

# checking which k to choose
wss <- sapply(2:10, function(k){
  kmeans(pcs$x[,1:10], centers=k)$tot.withinss
})

plot(2:10, wss, type="b", pch=19,
     xlab="k", ylab="Total within-cluster sum of squares")

# which PCs explain variance
plot(pcs$sdev^2 / sum(pcs$sdev^2), type="b",
     xlab="PC", ylab="Variance explained")


# Quick check plots
p1 <- ggplot(df, aes(x = PC1, y = PC2, col = clusters)) +
  geom_point(size = 0.7, alpha = 0.7) +
  ggtitle("Clusters in PCA space (PC1 vs PC2)")

p2 <- ggplot(df, aes(x = x, y = y, col = clusters)) +
  geom_point(size = 0.8, alpha = 0.8) +
  coord_fixed() +
  ggtitle("Clusters in tissue space")


# Identify the same cell-type using the same marker logic (Aqp1 + Slc34a1)
markers <- c("Aqp1", "Slc34a1")
present_markers <- markers[markers %in% colnames(mat)]
print(paste("Markers present:", paste(present_markers, collapse = ", ")))

if (length(present_markers) == 0) {
  stop("Neither Aqp1 nor Slc34a1 is present in this dataset's gene columns.")
}



# scoring each cluster by mean expression of the markers
cluster_scores <- sapply(levels(clusters), function(cl) {
  cells <- names(clusters)[clusters == cl]
  mean(rowMeans(mat[cells, present_markers, drop = FALSE]))
})

cluster_interest <- as.integer(names(which.max(cluster_scores)))
print(cluster_scores)
print(paste("Selected cluster_interest =", cluster_interest))

cluster_cells <- names(clusters)[clusters == cluster_interest]
other_cells   <- names(clusters)[clusters != cluster_interest]


# Differential expression (Wilcoxon, greater) for the selected cluster
p_wilcox <- sapply(colnames(mat), function(gene) {
  x1 <- mat[cluster_cells, gene]
  x2 <- mat[other_cells, gene]
  wilcox.test(x1, x2, alternative = "greater")$p.value
})

eff_median <- sapply(colnames(mat), function(gene) {
  median(mat[cluster_cells, gene]) - median(mat[other_cells, gene])
})

de <- data.frame(
  gene = colnames(mat),
  p = p_wilcox,
  neglog10p = -log10(p_wilcox),
  med_diff = eff_median
) %>% arrange(p)

print(head(de, 30))


# Top DE genes bar plot
top_n <- 15
de_top <- de %>% slice(1:top_n) %>% mutate(gene = factor(gene, levels = rev(gene)))

p3 <- ggplot(de_top, aes(x = gene, y = neglog10p)) +
  geom_col() +
  coord_flip() +
  ggtitle(paste0("Cluster ", cluster_interest, " DE genes")) +
  ylab("-log10(p)") + xlab("Gene")


# Marker expression panels 
plot_markers <- present_markers  

df_mark <- df %>%
  mutate(id = rownames(df)) %>%
  select(id, x, y, PC1, PC2, clusters)

# long-format marker values for faceting
marker_long <- do.call(rbind, lapply(plot_markers, function(g) {
  data.frame(
    id = rownames(mat),
    marker = g,
    gene_expr = mat[, g]
  )
}))

df_gene <- left_join(df_mark, marker_long, by = "id")

p4 <- ggplot(df_gene, aes(x = PC1, y = PC2, col = gene_expr)) +
  geom_point(size = 0.7, alpha = 0.7) +
  facet_wrap(~ marker) +
  ggtitle("Marker expression in PCA space")
#p4
p5 <- ggplot(df_gene, aes(x = x, y = y, col = gene_expr)) +
  geom_point(size = 0.8, alpha = 0.8) +
  coord_fixed() +
  facet_wrap(~ marker) +
  ggtitle("Marker expression in tissue space")
#p5
```
