###Summary 
Figure Description:
This figure presents a comparative analysis of mouse kidney spatial transcriptomics data collected using two platforms: Xenium (single-cell resolution) and Visium (multi-cellular spot resolution). The goal was to identify and characterize Aqp2+ collecting duct principal cells across three analyses: STdeconvolve deconvolution, and K-means clustering form HW3 and 4.
The top panel shows STdeconvolve applied to the Visium dataset using Latent Dirichlet Allocation (LDA) with K=8 topics, visualized using scatterbar. Each stacked bar represents one Visium spot positioned at its physical tissue coordinate, with the red segment (Topic 7) representing the estimated proportion of Aqp2+ collecting duct principal cells. All other topics are shown in grey. Topic 7 was identified as the Aqp2+ topic because it had the highest Aqp2 gene loading across all 8 topics. The scatterbar shows that red segments are concentrated in the lower inner region of the tissue, corresponding to the renal inner medulla and papilla where collecting duct principal cells are known to reside.
The second row shows K-means clustering on both platforms in physical space. The third row highlights the Aqp2+ population specifically: on Visium, this is the continuous proportion of Topic 7 per spot from STdeconvolve, and on Xenium, this is the K-means cluster with the highest mean Aqp2 expression (Cluster 3). Both show Aqp2+ cells concentrated in the same inner medullary region of the kidney, consistent with known kidney anatomy.
The bottom row shows the top 10 differentially upregulated genes for the Aqp2+ population on each platform, computed using Wilcoxon tests comparing the Aqp2+ cluster/topic to all other cells.
Comparison of Results Across HW3, HW4, and EC2:
The 3 analyses are largely consistent. All 3 methods identify Aqp2+ cells in the same spatial location (the inner medulla and papilla of the kidney). This confirms that spatial signal is robust across platforms and methods.
The Xenium DE genes (Slc4a9, Slc26a4, Foxi1, Aqp6, Slc4a1, Atp6v1c2, Aqp2, Rhbg, Muc20, Dmrt2) are highly specific markers of collecting duct and intercalated cells, which makes sense given that Xenium has single-cell resolution and can cleanly separate these subtypes. The Visium K-means DE genes (Slc14a2, Nccrp1, Pcsk6, Cck, Fst, Scel, Gsdmc3, Ptpm2, Wnt7b, Aldoc) are different because each Visium spot contains multiple cell types blended together, so the cluster signal reflects the dominant cell type mix in that region rather than a pure cell type signature.
A key difference between STdeconvolve and K-means is that STdeconvolve estimates a proportion of each cell type per spot rather than forcing a hard assignment. This is visible in the Aqp2+ proportion map since spots near the boundary of the inner medulla show intermediate red values, meaning they contain a mixture of Aqp2+ and other cell types. K-means cannot capture this and would assign the whole spot to one cluster. However, STdeconvolve still operates at spot resolution and cannot distinguish the individual intercalated cell subtypes (marked by Foxi1, Slc26a4, Rhbg) that Xenium resolves at single-cell level.





```r

library(ggplot2)
library(patchwork)
library(topicmodels)
library(slam)
library(scatterbar)
library(STdeconvolve)
library(dplyr)


#load Xenium 
data_x <- read.csv("~/Desktop/Xenium-IRI-ShamR_matrix.csv.gz")
pos_x  <- data_x[, c("x", "y")]
rownames(pos_x) <- data_x$X
gexp_x <- data_x[, (which(colnames(data_x) == "y") + 1):ncol(data_x)]
rownames(gexp_x) <- data_x$X
mat_x  <- log10(gexp_x + 1)

#load Visium
data_v <- read.csv("~/Desktop/Visium-IRI-ShamR_matrix.csv.gz")
pos_v  <- data_v[, c("x", "y")]
rownames(pos_v) <- data_v$X
gexp_v <- data_v[, (which(colnames(data_v) == "y") + 1):ncol(data_v)]
rownames(gexp_v) <- data_v$X
mat_v  <- log10(gexp_v + 1)

#kmeans 
set.seed(42)
com_x <- as.factor(kmeans(scale(mat_x), centers = 10)$cluster)
names(com_x) <- rownames(gexp_x)

set.seed(42)
com_v <- as.factor(kmeans(scale(mat_v[, apply(mat_v, 2, sd) > 0]),
                          centers = 6)$cluster)
names(com_v) <- rownames(gexp_v)

#find Aqp2+ cluster 
aqp2_clust_x <- which.max(tapply(gexp_x[, "Aqp2"], com_x, mean))
aqp2_clust_v <- which.max(tapply(gexp_v[, "Aqp2"], com_v, mean))

#STdeconvolve
counts_v <- round(as.matrix(gexp_v))
pct<- colMeans(counts_v > 0)
counts_filt <- counts_v[, pct >= 0.05 & pct <= 0.95]
dtm<- slam::as.simple_triplet_matrix(counts_filt)

set.seed(42)
lda <- topicmodels::LDA(dtm, k = 8, method = "Gibbs",control = list(seed = 42, burnin = 100,iter  = 500, thin  = 10))

post  <- topicmodels::posterior(lda)
beta  <- post$terms   # topics x genes
theta <- post$topics  # spots x topics

#find Aqp2+ topic
aqp2_topic <- which.max(beta[, "Aqp2"])

colnames(theta) <- paste0("Topic", 1:8)

# STcompare: top genes enriched in Aqp2+ topic vs all others ─
b1  <- beta[aqp2_topic, ]
b2  <- colMeans(beta[-aqp2_topic, ])
stc <- sort(b1 / (b2 + 1e-10), decreasing = TRUE)[1:10]
print(stc)

# DE analysis
cells_in_x  <- names(com_x)[com_x == aqp2_clust_x]
cells_out_x <- names(com_x)[com_x != aqp2_clust_x]
fc_x  <- log2((colMeans(gexp_x[cells_in_x,  ]) + 0.01) /
                  (colMeans(gexp_x[cells_out_x, ]) + 0.01))
top10_x <- names(sort(fc_x, decreasing = TRUE))[1:10]

cells_in_v  <- names(com_v)[com_v == aqp2_clust_v]
cells_out_v <- names(com_v)[com_v != aqp2_clust_v]
fc_v  <- log2((colMeans(gexp_v[cells_in_v,  ]) + 0.01) /
                  (colMeans(gexp_v[cells_out_v, ]) + 0.01))
top10_v <- names(sort(fc_v, decreasing = TRUE))[1:10]

# ────────────────

#scatterbar
#colors chosen from the color brewer
topic_cols <- c("#e74c3c","#3498db","#2ecc71","#f39c12",
                "#9b59b6","#1abc9c","#e67e22","#95a5a6")
names(topic_cols) <- paste0("Topic", 1:8)

p1 <- scatterbar(theta[rownames(pos_v), ],
                 pos_v[, c("x","y")],
                 colors = topic_cols) +
  ggtitle("STdeconvolve: cell-type proportions (Visium)") +
  coord_fixed() + theme_minimal()

## kmeans Visium
p2 <- ggplot(data.frame(pos_v, cluster = com_v), aes(x, y, col = cluster)) +
  geom_point(size = 0.3, alpha = 0.6) +
  ggtitle("Visium K-means (K=6)") + coord_fixed() 

## kmeans Xenium
p3 <- ggplot(data.frame(pos_x, cluster = com_x), aes(x, y, col = cluster)) +
  geom_point(size = 0.15, alpha = 0.3) +
  ggtitle("Xenium K-means (K=10)") + coord_fixed() + 

## Aqp2+ on Visium
pos_v$aqp2 <- theta[rownames(pos_v), aqp2_topic]
p4 <- ggplot(pos_v, aes(x, y, col = aqp2)) +
  geom_point(size = 0.3, alpha = 0.6) +
  scale_color_gradient(low = "lightblue", high = "red", name = "Aqp2+") +
  ggtitle("Aqp2+ proportion: Visium") +
  coord_fixed() + theme_minimal()

## 5. Aqp2+ on Xenium
pos_x$aqp2 <- com_x == aqp2_clust_x
p5 <- ggplot(pos_x, aes(x, y, col = aqp2)) +
  geom_point(size = 0.15, alpha = 0.6) +
  scale_color_manual(values = c("FALSE" = "lightblue", "TRUE" = "red"),
                     labels = c("FALSE" = "other", "TRUE" = "Aqp2+")) +
  ggtitle("Aqp2+ cluster: Xenium") +
  coord_fixed() + theme_minimal()

## DE genes Xenium
p6 <- ggplot(data.frame(gene = top10_x, fc = fc_x[top10_x]),
             aes(fc, reorder(gene, fc))) +
  geom_col(fill = "steelblue") +
  labs(title = paste("Top 10 DE genes: Xenium cluster", aqp2_clust_x),
       x = "log2 FC", y = "") + theme_minimal()

##DE genes Visium
p7 <- ggplot(data.frame(gene = top10_v, fc = fc_v[top10_v]),
             aes(fc, reorder(gene, fc))) +
  geom_col(fill = "steelblue") +
  labs(title = paste("Top 10 DE genes: Visium cluster", aqp2_clust_v),
       x = "log2 FC", y = "") + theme_minimal()

#assemble
library(gridExtra)
png("ec2_figure.png", width = 2200, height = 2200, res = 150)
grid.arrange(
  p1 + labs(x = "x", y = "y"),
  p2, p3,
  p4, p5,
  p6, p7,
  layout_matrix = rbind(c(1, 1, 1),
                        c(2, 2, 3),
                        c(4, 4, 5),
                        c(6, 7, 8)))



```
