### Discussion of solution

I applied STdeconvolve to the Visium dataset and selected K = 7 latent topics based on perplexity stabilization. In contrast to k-means clustering (k = 5),
which assigns each spot to a single discrete cluster, deconvolution revealed graded mixtures of cell-type proportions across spatial locations. Topic 3 showed 
the highest spatial variability and was enriched for canonical tubular epithelial markers including Umod, Slc12a1, Egf, and Aqp1. Spatial visualization confirmed 
coherent anatomical localization. Cross-platform comparison demonstrated that the same marker genes were strongly expressed at single-cell resolution in the Xenium 
dataset, and differential expression analyses across both platforms recovered consistent epithelial gene programs. These results illustrate how deconvolution captures 
biologically meaningful compositional structure that is obscured by hard clustering approaches.

REFERENCE: Cheval, L., et al. (2012). Atlas of gene expression in the mouse kidney: new features of glomerular parietal cells. American Journal of Physiology – Renal 
Physiology, 303(3), F342–F354.

NOTE: Please see the additional graphs/figures and tables the code produces that for supporting material. These additional figures were (e.g scree plot, fitted model Ks vs deconvolve cell/perplexitiy) were not included in the final multi-panel figure

# Normalizing counts to log10(CPM + 1) (same pattern used in class/HWs)
log_cpm10 <- function(count_mat) {
  tot <- rowSums(count_mat)
  # avoid divide-by-zero
  tot[tot == 0] <- 1
  log10(count_mat / tot * 1e6 + 1)
}

# Simple differential expression: cluster/topic vs rest, using Wilcoxon
# KINDLY NOTE: Returns table with logFC (difference of means) + p-values
de_wilcox <- function(mat_log, group_labels, group_of_interest) {
  stopifnot(nrow(mat_log) == length(group_labels))
  group_labels <- as.character(group_labels)
  
  idx_in <- which(group_labels == group_of_interest)
  idx_out <- which(group_labels != group_of_interest)
  
  # mean difference as effect size
  mu_in <- colMeans(mat_log[idx_in, , drop = FALSE])
  mu_out <- colMeans(mat_log[idx_out, , drop = FALSE])
  logFC <- mu_in - mu_out
  
  # p-values (Wilcoxon)
  pvals <- sapply(seq_len(ncol(mat_log)), function(j) {
    suppressWarnings(wilcox.test(mat_log[idx_in, j], mat_log[idx_out, j])$p.value)
  })
  
  tibble(
    gene = colnames(mat_log),
    logFC = as.numeric(logFC),
    pval = as.numeric(pvals),
    padj = p.adjust(pvals, method = "BH")
  ) %>%
    arrange(padj, desc(logFC))
}

# Building a barplot of top DE genes
plot_de_bar <- function(de_tbl, title, top_n = 12) {
  top <- de_tbl %>%
    filter(is.finite(logFC), is.finite(padj)) %>%
    arrange(padj, desc(logFC)) %>%
    slice_head(n = top_n) %>%
    mutate(gene = factor(gene, levels = rev(gene)))
  
  ggplot(top, aes(x = gene, y = logFC)) +
    geom_col() +
    coord_flip() +
    labs(title = title, x = NULL, y = "Mean(logCPM) difference (in - out)") +
    theme_bw(base_size = 11)
}

# Plotting a continuous value on tissue coordinates
plot_spatial_continuous <- function(pos_df, value_vec, title, pt_size = 0.8) {
  df <- pos_df %>%
    mutate(value = value_vec)
  
  ggplot(df, aes(x = x, y = y, col = value)) +
    geom_point(size = pt_size) +
    coord_fixed() +
    theme_bw(base_size = 11) +
    labs(title = title, col = NULL)
}

# Plotting discrete clusters on tissue coordinates
plot_spatial_discrete <- function(pos_df, clust, title, pt_size = 0.8) {
  df <- pos_df %>% mutate(cluster = as.factor(clust))
  ggplot(df, aes(x = x, y = y, col = cluster)) +
    geom_point(size = pt_size) +
    coord_fixed() +
    theme_bw(base_size = 11) +
    labs(title = title, col = "Cluster")
}
